## 二、补充优化（用户方案未提及）

| # | 优化点 | 原因 |
|---|--------|------|
| 1 | 写文件加 `OpenMode.TRUNC` | 新内容比旧内容短时，残留旧数据导致 JSON 损坏 |
| 2 | 空文件处理 | 文件存在但 0 字节时，`JSON.parse('')` 会抛异常 |
| 3 | `statSync` 用 `filePath` | 不依赖 `file.path`，更可靠 |

## 三、最终实现代码

```typescript
import { fileIo } from '@kit.CoreFileKit';
import { buffer } from '@kit.ArkTS';
import { common } from '@kit.AbilityKit';
import { BubbleTree, BubbleNode, ChatMessage } from '../common/ChatModel';

// ========== 1. 响应式树 → 纯对象（剥离 @ObservedV2 代理） ==========
export function treeToPlainObj(tree: BubbleTree): object {
  const plainNodes: object[] = [];
  for (let i = 0; i < tree.nodeList.length; i++) {
    const node = tree.nodeList[i];
    // 逐个提取 ChatMessage 属性
    const plainMsgs: object[] = [];
    for (let j = 0; j < node.messages.length; j++) {
      const msg = node.messages[j];
      plainMsgs.push({
        messageId: msg.messageId,
        role: msg.role,
        content: msg.content
      });
    }
    // 逐个提取 BubbleNode 属性
    plainNodes.push({
      nodeID: node.nodeID,
      parentID: node.parentID,
      x: node.x, y: node.y,
      parentX: node.parentX, parentY: node.parentY,
      messages: plainMsgs
    });
  }
  return { version: 1, nodeList: plainNodes };
}

// ========== 2. 纯对象 → 重建响应式 BubbleTree ==========
export function plainObjToTree(obj: any): BubbleTree {
  // 数据校验
  if (!obj || !obj.nodeList || !Array.isArray(obj.nodeList)) {
    return new BubbleTree([]);
  }
  const nodeList: BubbleNode[] = [];
  for (let i = 0; i < obj.nodeList.length; i++) {
    const n = obj.nodeList[i];
    // 重建 ChatMessage 数组
    const messages: ChatMessage[] = [];
    if (n.messages && Array.isArray(n.messages)) {
      for (let j = 0; j < n.messages.length; j++) {
        const m = n.messages[j];
        messages.push(new ChatMessage(m.messageId, m.role, m.content));
      }
    }
    // 重建 BubbleNode，兜底默认值
    nodeList.push(new BubbleNode(
      n.nodeID, n.parentID, messages,
      n.x, n.y,
      n.parentX || 0, n.parentY || 0
    ));
  }
  return new BubbleTree(nodeList);
}

// ========== 3. 保存到沙盒 ==========
export async function saveBubbleTree(tree: BubbleTree): Promise<boolean> {
  try {
    const plain = treeToPlainObj(tree);           // 代理 → 纯对象
    const json = JSON.stringify(plain);           // 序列化
    const ctx = getContext() as common.UIAbilityContext;
    const filePath = ctx.filesDir + '/bubbleTreeData.json';
    // TRUNC: 清空旧内容，防止新内容更短时残留
    const file = fileIo.openSync(filePath,
      fileIo.OpenMode.READ_WRITE | fileIo.OpenMode.CREATE | fileIo.OpenMode.TRUNC);
    fileIo.writeSync(file.fd, json);              // 写入
    fileIo.closeSync(file);
    return true;
  } catch (e) {
    console.error(`[TreeFileUtil] save failed: ${e}`);
    return false;
  }
}

// ========== 4. 从沙盒加载 ==========
export async function loadBubbleTree(): Promise<BubbleTree | null> {
  try {
    const ctx = getContext() as common.UIAbilityContext;
    const filePath = ctx.filesDir + '/bubbleTreeData.json';
    // 判断文件是否存在
    try {
      fileIo.accessSync(filePath);
    } catch {
      return null; // 文件不存在
    }
    // 读取文件
    const file = fileIo.openSync(filePath, fileIo.OpenMode.READ_ONLY);
    const stat = fileIo.statSync(filePath);
    if (stat.size === 0) {
      fileIo.closeSync(file);
      return null; // 空文件
    }
    const arrayBuffer = new ArrayBuffer(stat.size);
    fileIo.readSync(file.fd, arrayBuffer);
    fileIo.closeSync(file);
    // 解析 JSON
    const text = buffer.from(arrayBuffer).toString();
    const obj = JSON.parse(text);
    // 重建响应式实例
    return plainObjToTree(obj);
  } catch (e) {
    console.error(`[TreeFileUtil] load failed: ${e}`);
    return null;
  }
}
```

## 四、接入 BBTreeCanvas 使用示例

## 五、注意事项

1. **代理对象序列化坑**：`JSON.stringify(bubbleNode)` 返回 `{}` 或丢失 `@Trace` 数据，必须用 `treeToPlainObj` 逐属性提取
2. **不要重复 connect**：组件已通过 `AppStorageV2.connect` 拿到 `store`，加载后直接 `this.store.nodeList = tree.nodeList`
3. **防抖**：只在 `onActionEnd` 保存一次，不要在 `onActionUpdate` 高频保存
4. **TRUNC 模式**：写文件时加 `OpenMode.TRUNC`，防止新内容更短时残留旧数据
5. **沙盒文件调试**：路径 `/data/app/.../files/bubbleTreeData.json`，DevEco Device File Browser 可查看
